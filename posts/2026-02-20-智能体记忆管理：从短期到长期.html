1. 为什么需要记忆

想象一个智能助手，每次对话都忘记你是谁、你们聊过什么。这样的体验显然 unacceptable。记忆是智能体从"工具"进化为"伙伴"的关键。

1.1 无记忆的问题

场景 无记忆的表现 有记忆的表现

执行任务 重复询问已给信息 自动使用之前确认的偏好
学习改进 反复犯同样的错误 根据反馈调整策略
情感连接 冷冰冰的问答 记得用户的生日、喜好

2. 记忆类型：短期vs长期

2.1 短期记忆（Short-term Memory）

定义：当前对话窗口内的上下文信息

特点：

  • 时效性强，会话结束即失效
  • 容量有限，受模型上下文窗口限制
  • 读写速度快，直接在内存中操作

典型实现：

# 简化的短期记忆
conversation_history = [
    {"role": "user", "content": "你好"},
    {"role": "assistant", "content": "你好！有什么我可以帮助你的？"},
    {"role": "user", "content": "我想订一张去北京的机票"},
]

2.2 长期记忆（Long-term Memory）

定义：跨越多个会话持久化存储的信息

特点：

  • 持久化存储，会话间保留
  • 容量大，可存储数月甚至数年的数据
  • 需要检索机制快速定位相关信息

分类：

类型 内容 示例
事实记忆 客观信息 用户的职位、公司
情景记忆 经历的事件 "上次会议讨论了产品发布"
程序记忆 技能和流程 用户偏好的报告格式

2.3 记忆对比表

维度 短期记忆 长期记忆
存储位置 内存/上下文窗口 数据库/向量存储
保留时间 单次会话 永久
容量 有限（4K-128K tokens） 理论上无限
访问速度 极快 依赖检索算法
实现难度 简单 中等

3.2 Token-aware 截断

考虑到模型上下文限制，按token截断更精确：

import tiktoken

class TokenAwareMemory:
    def __init__(self, max_tokens=4000, model="gpt-3.5-turbo"):
        self.messages = []
        self.max_tokens = max_tokens
        self.encoding = tiktoken.encoding_for_model(model)
    
    def count_tokens(self, text):
        return len(self.encoding.encode(text))
    
    def add(self, role, content):
        self.messages.append({"role": role, "content": content})
        self._truncate_if_needed()
    
    def _truncate_if_needed(self):
        """当超出token限制时，从最早的消息开始删除"""
        total_tokens = sum(
            self.count_tokens(m["content"]) for m in self.messages
        )
        
        while total_tokens > self.max_tokens and len(self.messages) > 2:
            # 保留system消息，删除最早的user/assistant对
            removed = self.messages.pop(1)  # 跳过system消息
            total_tokens -= self.count_tokens(removed["content"])

3.3 摘要压缩

当历史过长时，可以压缩为摘要：

async def compress_history(self, llm_client):
    """将早期对话压缩为摘要"""
    if len(self.messages) < 10:
        return
    
    # 保留最近5条，压缩前面的
    to_compress = self.messages[:-5]
    recent = self.messages[-5:]
    
    # 生成摘要
    summary_prompt = f"""
    请总结以下对话的关键信息（限制100字）：
    {to_compress}
    """
    summary = await llm_client.complete(summary_prompt)
    
    # 重置为：系统摘要 + 最近消息
    self.messages = [
        {"role": "system", "content": f"历史对话摘要：{summary}"}
    ] + recent

4.1 SQLite 简单实现

import sqlite3
from datetime import datetime

class LongTermMemory:
    def __init__(self, db_path="memory.db"):
        self.db_path = db_path
        self._init_db()
    
    def _init_db(self):
        """初始化数据库表"""
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                CREATE TABLE IF NOT EXISTS memories (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    user_id TEXT NOT NULL,
                    category TEXT,
                    content TEXT NOT NULL,
                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                )
            """)
            conn.execute("""
                CREATE TABLE IF NOT EXISTS conversations (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    user_id TEXT NOT NULL,
                    summary TEXT,
                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                )
            """)
    
    def save_fact(self, user_id, category, content):
        """保存事实记忆"""
        with sqlite3.connect(self.db_path) as conn:
            # 检查是否已存在
            existing = conn.execute(
                "SELECT id FROM memories WHERE user_id=? AND category=?",
                (user_id, category)
            ).fetchone()
            
            if existing:
                # 更新
                conn.execute(
                    "UPDATE memories SET content=?, updated_at=? WHERE id=?",
                    (content, datetime.now(), existing[0])
                )
            else:
                # 插入
                conn.execute(
                    "INSERT INTO memories (user_id, category, content) VALUES (?, ?, ?)",
                    (user_id, category, content)
                )
            conn.commit()
    
    def get_facts(self, user_id, category=None):
        """获取用户的事实记忆"""
        with sqlite3.connect(self.db_path) as conn:
            if category:
                cursor = conn.execute(
                    "SELECT category, content FROM memories WHERE user_id=? AND category=?",
                    (user_id, category)
                )
            else:
                cursor = conn.execute(
                    "SELECT category, content FROM memories WHERE user_id=?",
                    (user_id,)
                )
            return cursor.fetchall()

import chromadb

class VectorMemory:
    def __init__(self, collection_name="memories"):
        self.model = SentenceTransformer('paraphrase-multilingual-MiniLM-L12-v2')
        self.client = chromadb.PersistentClient(path="./chroma_db")
        self.collection = self.client.get_or_create_collection(collection_name)
    
    def add(self, text, metadata=None, id=None):
        """添加记忆到向量库"""
        embedding = self.model.encode(text).tolist()
        
        if id is None:
            id = f"mem_{datetime.now().timestamp()}"
        
        self.collection.add(
            embeddings=[embedding],
            documents=[text],
            metadatas=[metadata or {}],
            ids=[id]
        )
    
    def search(self, query, n_results=5):
        """语义搜索相关记忆"""
        query_embedding = self.model.encode(query).tolist()
        
        results = self.collection.query(
            query_embeddings=[query_embedding],
            n_results=n_results
        )
        
        return results["documents"][0]
    
    def get_relevant_context(self, query, user_id=None):
        """获取与查询相关的上下文"""

        return self.search(query)

class MemoryRetriever:
    def __init__(self, short_term, long_term):
        self.short_term = short_term
        self.long_term = long_term
    
    async def build_prompt_context(self, user_input, user_id):
        """构建完整的提示上下文"""
        context_parts = []
        
        # 1. 系统指令
        context_parts.append({
            "role": "system",
            "content": "你是一个有帮助的AI助手。以下是相关信息："
        })
        
        # 2. 长期记忆（用户偏好、事实）
        facts = self.long_term.get_facts(user_id)
        if facts:
            facts_text = "\n".join([f"- {cat}: {val}" for cat, val in facts])
            context_parts.append({
                "role": "system",
                "content": f"用户已知信息：\n{facts_text}"
            })
        
        # 3. 相关历史对话（向量检索）
        if hasattr(self.long_term, 'get_relevant_context'):
            relevant = self.long_term.get_relevant_context(user_input, user_id)
            if relevant:
                context_parts.append({
                    "role": "system",
                    "content": f"相关历史：{' '.join(relevant)}"
                })
        
        # 4. 短期记忆（最近对话）
        context_parts.extend(self.short_term.get_context())
        
        # 5. 当前输入
        context_parts.append({"role": "user", "content": user_input})
        
        return context_parts

5. 记忆压缩策略

5.1 选择性记忆

不是所有信息都值得长期存储：

class MemoryFilter:
    """决定哪些信息值得记住"""
    
    IMPORTANT_CATEGORIES = [
        "user_preference",    # 用户偏好
        "personal_info",      # 个人信息（职位、公司等）
        "learned_fact",       # 学到的事实
        "task_outcome",       # 任务结果
    ]
    
    FORGETTABLE_PATTERNS = [
        "你好", "谢谢", "不客气",  # 礼貌用语
        "稍等", "明白", "好的",     # 确认词
    ]
    
    def should_remember(self, content, category=None):
        # 检查是否属于重要类别
        if category in self.IMPORTANT_CATEGORIES:
            return True
        
        # 检查是否只是礼貌用语
        if any(p in content for p in self.FORGETTABLE_PATTERNS):
            return False
        
        # 内容长度检查（太短的信息价值低）
        if len(content) < 10:
            return False
        
        return True

5.2 分层存储

class TieredMemory:
    """三层存储架构"""
    
    def __init__(self):
        self.l1_cache = {}      # L1: 内存缓存（最近1小时）
        self.l2_redis = None    # L2: Redis（最近24小时）
        self.l3_db = None       # L3: 数据库（永久）
    
    async def get(self, key):
        # 先查L1
        if key in self.l1_cache:
            return self.l1_cache[key]
        
        # 再查L2
        if self.l2_redis:
            value = await self.l2_redis.get(key)
            if value:
                self.l1_cache[key] = value  # 回填L1
                return value
        
        # 最后查L3
        if self.l3_db:
            value = await self.l3_db.get(key)
            if value:
                # 回填L1、L2
                self.l1_cache[key] = value
                if self.l2_redis:
                    await self.l2_redis.set(key, value, ex=3600)
                return value
        
        return None

5.3 记忆衰减

模拟人类记忆的遗忘曲线：

import math
from datetime import datetime, timedelta

class DecayingMemory:
    """基于时间的记忆衰减"""
    
    def __init__(self, half_life_days=30):
        self.half_life = half_life_days
    
    def calculate_relevance(self, memory_timestamp, importance=1.0):
        """
        计算记忆的相关性分数
        使用指数衰减模型
        """
        age_days = (datetime.now() - memory_timestamp).days
        decay_factor = math.exp(-0.693 * age_days / self.half_life)
        return importance * decay_factor
    
    def should_prune(self, memory_timestamp, importance=1.0, threshold=0.1):
        """判断是否应该删除这条记忆"""
        relevance = self.calculate_relevance(memory_timestamp, importance)
        return relevance < threshold

import asyncio
from typing import List, Dict, Any

class PersonalAssistant:
    def __init__(self, user_id: str, llm_client):
        self.user_id = user_id
        self.llm = llm_client
        
        # 初始化记忆系统
        self.short_term = TokenAwareMemory(max_tokens=3000)
        self.long_term = SQLiteMemory(db_path=f"user_{user_id}.db")
        self.vector_mem = VectorMemory(collection_name=f"user_{user_id}")
        self.retriever = MemoryRetriever(self.short_term, self.vector_mem)
    
    async def chat(self, user_input: str) -> str:
        """主对话函数"""
        
        # 1. 提取重要信息
        await self._extract_and_store_facts(user_input)
        
        # 2. 构建上下文
        context = await self.retriever.build_prompt_context(
            user_input, self.user_id
        )
        
        # 3. 调用LLM
        response = await self.llm.complete(context)
        
        # 4. 更新短期记忆
        self.short_term.add("user", user_input)
        self.short_term.add("assistant", response)
        
        # 5. 保存到向量库用于未来检索
        self.vector_mem.add(
            text=f"User: {user_input}\nAssistant: {response}",
            metadata={"timestamp": datetime.now().isoformat()}
        )
        
        return response
    
    async def _extract_and_store_facts(self, text: str):
        """从对话中提取事实并存储"""
        
        # 使用LLM提取关键信息
        extraction_prompt = f"""
        从以下对话中提取关键事实（如用户偏好、个人信息等）。
        只提取重要、持久的信息。如果没有则返回"无"。
        
        对话：{text}
        
        格式：类别: 内容
        """
        
        extracted = await self.llm.complete(extraction_prompt)
        
        # 解析并存储
        for line in extracted.split("\n"):
            if ":" in line and line.strip() != "无":
                category, content = line.split(":", 1)
                self.long_term.save_fact(
                    self.user_id,
                    category.strip(),
                    content.strip()
                )
    
    def get_user_profile(self) -> Dict[str, Any]:
        """获取用户画像"""
        facts = self.long_term.get_facts(self.user_id)
        return {cat: content for cat, content in facts}
